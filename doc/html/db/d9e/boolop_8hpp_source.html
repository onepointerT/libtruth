31 static Type is( const std::string op ) {

35 static const std::string string( const BoolOperator::Type op ) {
36 if ( op == Type::AND ) return " && ";
37 else if ( op == Type::OR ) return " || ";
38 else if ( op == Type::IMPLIES ) return " >> ";
39 else if ( op == Type::NEGATION ) return " ! ";
40 else if ( op == Type::ANY ) return "{ &&, ||, >> }";
41 else return "";
42 }

44 static bool eval( const bool lhs, const BoolOperator::Type op, const bool rhs ) {
45 if ( op == Type::UNKNOWN ) return false;
46 else if ( op == Type::AND ) return lhs && rhs;
47 else if ( op == Type::OR ) return lhs || rhs;
48 else if ( op == Type::IMPLIES ) return !(lhs && !rhs) && rhs;
49 else if ( op == Type::NEGATION ) return !lhs;
50 else return false;
51 }
