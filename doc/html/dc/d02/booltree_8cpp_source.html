63TreeNode::node_children_pair_t* TreeNode::split() const {
64
65 BoolExpr::expr_pair_t exprpair
66 = this->m_expr.splitBeforeLastOpeningTermBracket();
67
68 if ( exprpair.first.first == nullptr && exprpair.first.second == nullptr )
69 exprpair = this->m_expr.splitAtFirstOperator();
70
71 TreeNode* node1 = new TreeNode( *exprpair.first.first, exprpair.second, true );
72 TreeNode* node2 = new TreeNode( *exprpair.first.second, exprpair.second );
73
74 node1->neighbour_right = node2;
75 node2->neighbour_left = node1;
76
77 return new node_children_pair_t{ node1, node2 };
78}

112void Tree::build_node( node_t* node ) {
113
114 BoolExpr::expr_pair_t ept;
115 if ( node->content->expr().contains(")") ) {
116 ept = node->content->expr().splitBehindFirstClosingTermBracket();
117 } else {
118 ept = node->content->expr().splitAtFirstOperator();
119 }
120
121 if ( ept.first.first == nullptr || ept.first.second == nullptr ) {
122 this->build_leafs_of_node( node );
123 }
124
125 node->content->op = ept.second;
126 TreeNode* tn1 = new TreeNode( *(ept.first.first), ept.second, true );
127 TreeNode* tn2 = new TreeNode( *(ept.first.second), ept.second );
128 tn1->neighbour_right = tn2;
129 tn2->neighbour_left = tn1;
130
131 node_t* snode1 = new node_t( *tn1, node );
132 node_t* snode2 = new node_t( *tn2, node );
133
134 node->add_sub( *snode1 );
135 node->add_sub( *snode2 );
136
137 for ( node_t* cnode : node->subs ) {
138 this->build_node( node );
139 }
140}

142Tree::node_t* Tree::build( const BoolExpr& logical_expr, node_t* node ) {
143
144 node_t* cnode = node;
145 if ( this->root == nullptr ) {
146 this->build_root( logical_expr );
147 cnode = this->root;
148 this->build_node( this->root );
149 } else {
150 this->build_node( node );
151 }
152
153 return cnode;
154}

201BoolValue Tree::eval( const leaf_t* leaf, const BoolDict<BoolValue>* bd ) const {
202 if ( leaf->content->result != BoolValue::None && leaf->content->result != BoolValue::Unknown ) {
203 return leaf->content->result;
204 }
205
206 bool leaf_result = leaf->content->term->eval( *bd );
207 leaf->content->result.value = leaf_result;
208 leaf->parent->content->terms_truth().first = leaf->content->result;
209 return leaf_result;
210}

213BoolValue Tree::eval( const node_t* node, const BoolDict<BoolValue>* bd ) const {
214
215 BoolValue lchld, rchld, thisnode = BoolValue::Unknown;
216 if ( node->subs.size() > 0 ) {
217 node_t* cnode = node->subs[0];
218 lchld = this->eval( cnode, bd );
219 }
220 if ( node->subs.size() > 1 ) {
221 node_t* cnode = node->subs[1];
222 rchld = this->eval( cnode, bd );
223 }
224 if ( node->leafs.size() > 0 ) {
225 for ( const leaf_t* leaf : node->leafs ) {
226 this->eval( leaf, bd );
227 }
228 thisnode = BoolOperator::eval(
229 node->content->terms_truth().first
230 , node->content->op
231 , node->content->terms_truth().second
232 );
233 }
234
235 if ( lchld != BoolValue::Unknown && rchld != BoolValue::Unknown )
236 return BoolOperator::eval( lchld, node->content->op, rchld );
237 else if ( lchld != BoolValue::Unknown )
238 return BoolOperator::eval( lchld, node->content->op, node->content->terms_truth().second );
239 else if ( rchld != BoolValue::Unknown )
240 return BoolOperator::eval( node->content->terms_truth().first, node->content->op, rchld );
241 else if ( thisnode != BoolValue::Unknown && thisnode != BoolValue::None )
242 return thisnode;
243 return BoolValue::None;
244}
