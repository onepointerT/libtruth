40const std::vector< AlgebraicSmallTerm* >& small_terms( const BoolExpr& expr ) {
41
42 std::vector< AlgebraicSmallTerm* >* vast = new std::vector< AlgebraicSmallTerm* >();
43
44 BoolType::small_term_queue_t* stq = BoolType::find_operator_in_expr( expr );
45
46 for ( const oneptr::queue_element< BoolType::small_term_t >* qe : { stq->front(), stq->back() } ) {
47 AlgebraicSmallTerm* ast = new AlgebraicSmallTerm( *(qe->get()) );
48 vast->push_back( ast );
49 }
50
51 return *vast;
52}

62BoolDict<BoolValue>& AlgebraicBind::fillDict( const bindval_t expr, BoolDict<BoolValue>& bd ) {
63 const std::vector< AlgebraicSmallTerm* >& left_expr_vec = small_terms( expr.first.first );
64 const std::vector< AlgebraicSmallTerm* >& right_expr_vec = small_terms( expr.first.second );
65
66 for ( AlgebraicSmallTerm* ast : left_expr_vec ) {
67 ast->addToDict( bd );

69 for ( AlgebraicSmallTerm* ast : right_expr_vec ) {
70 ast->addToDict( bd );

130BoolDict<BoolValue>& AlgebraicSmallTerm::fillDict( const varnames_t varnames, BoolDict<BoolValue>& bd ) {
131
132 bool has_var1 = false;
133 bool has_var2 = false;
134 for ( BoolDict<BoolValue>::const_iterator cit : { bd.begin(), bd.end() } ) {
135 if ( cit->first == varnames.first ) has_var1 = true;
136 if ( cit->first == varnames.second ) has_var2 = true;
137 }
138 if ( ! has_var1 ) bd.at( varnames.first ) = BoolValue::Unknown;
139 if ( ! has_var2 ) bd.at( varnames.second ) = BoolValue::Unknown;
140
141 return bd;
142}
