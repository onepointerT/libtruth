29BoolTerm BoolExpr::is() const {
30 if ( compare( 0, 1, "&" ) == 0 || compare( 0, 1, "|" ) == 0 )
31 return BoolTerm::OP;
32 else if ( compare( 0, 1, "(" ) == 0 ) return BoolTerm::LBRACKET;
33 else if ( compare( 0, 1, ")" ) == 0 ) return BoolTerm::RBRACKET;
34 else if ( compare( 0, 1, "!" ) == 0 ) return BoolTerm::NEGATION;
35 else if ( compare( 0, 1, " " ) == 0 && compare( length()-1, 1, " " ) == 0 )
36 return BoolTerm::VARIABLE;
37 else return BoolTerm::TERM;
38}
