40bool BoolType::boolval( const arguments_t args, const BoolOperator::Type op ) {
41
42 if ( op == BoolOperator::Type::AND ) {
43 for ( typename BoolType::arguments_iter_t val : args ) {
44 if ( val.second != True ) return false;
45 }
46 } else if ( op == BoolOperator::OR ) {
47 for ( typename BoolType::arguments_iter_t val : args ) {
48 if ( val.second != False ) return true;
49 }
50 }
51
52 return true;
53}

56BoolType::term_t BoolType::find_terms( expr_t expr, bool_expression_t& restexpr ) {
57
58 oneptr::queue< atomic_term_t >* atq = new oneptr::queue< atomic_term_t >();
59 atomic_terms_t* atx = new atomic_terms_t{ expr, atq };
60
61 bool term_begin = false;
62 std::string bte;
63 for ( char e : expr ) {
64 if ( e == ' ' && ! term_begin ) {
65 term_begin = true;
66 } else if ( term_begin && e == ' ' ) {
67 term_begin = false;
68 BoolExpr be( bte );
69 BoolTerm bt = be.is();
70 atq->push_back( { be, bt } );
71 bte = "";
72 } else {
73 bte += e;
74 }
75 }
76
77 restexpr.first = expr.find_prefix_before( bte );
78 restexpr.second = expr.find_suffix_behind( bte );
79
80 return { atx, BoolExpr( bte ).is() };
81}

84BoolType::expression_t BoolType::find_exprs( expr_t expr, bool_expression_t& restexpr ) {
85
86 std::string exprstr = expr;
87
88 oneptr::queue< atomic_expr_t >* aet
89 = new oneptr::queue< atomic_expr_t >();
90 BoolExpr reststring_brackets = exprstr;
91 size_t aet_size_before = 0;
92 while ( reststring_brackets.length() != 0
93 && aet_size_before < aet->size()
94 ) {
95 expression_t exp = BoolType::atomic_bracket_find( exprstr, 1 );
96 aet->emplace_back( exp.first );
97 reststring_brackets -= exp.second;
98 }
99
100 restexpr.first = expr.find_prefix_before( reststring_brackets );
101 restexpr.second = expr.find_suffix_behind( reststring_brackets );
102
103
104 atomic_expr_t* aex = new atomic_expr_t();
105 for ( oneptr::queue_element<atomic_expr_t>* qeae : { aet->front(), aet->back() } ) {
106
107 bool_expression_t bet{ "", "" };
108
109 for ( oneptr::queue_element<atomic_terms_t>* qeat : { (*qeae)()->front(), (*qeae)()->back() } ) {
110 aex->emplace_back( (*qeat)() );
111 }
112 }
113
114 BoolExpr evaluated_expr = ( expr - restexpr.first ) - restexpr.second;
115
116 return { aex, evaluated_expr };
117}

120BoolType::expr_vec_t BoolType::atomic_find( expr_t expr, size_t max_terms ) {
121
122 BoolExpr new_expr;
123 std::string exprstr = expr;
124 size_t epos = 0;
125 unsigned int tfound = 0;
126
127 expr_vec_t exprvec;
128
129 std::string estr = "";
130 for ( unsigned int e = epos; e < exprstr.length(); e = epos ) {
131 char ec = exprstr[e];
132
133 if ( tfound >= max_terms && max_terms > 0 ) break;
134 else if ( ec == '(' ) {
135 ++tfound;
136
137 expression_t bracket_expr = BoolType::atomic_bracket_find( exprstr.substr( epos ).c_str(), 1 );
138 estr += bracket_expr.second;
139
140 BoolExpr bexpr = estr;
141
142 exprvec.emplace_back( bool_expr_t{ bool_expression_t{ bexpr, expr }, truth::BRACKETTERM } );
143
144 if ( BoolType::atomic_expr_count( bracket_expr.second ) + tfound <= max_terms && max_terms > 0 ) {
145 expr_vec_t bexprvec = BoolType::atomic_find( bracket_expr.second, max_terms == 0 ? 0 : max_terms - tfound );
146
147 if ( bexprvec.size() > 0 ) exprvec.append_range( std::ranges::subrange{ bexprvec.begin(), bexprvec.end() } );
148 }
149
150 epos = exprstr.find_first_not_of( bracket_expr.second, e );
151
152 } else if ( ec == ' ' ) {
153
154 size_t vpos_end = exprstr.find_first_of( ' ', e+1 );
155
156 BoolExpr var_expr = exprstr.substr( e+1, vpos_end-e );
157 estr += var_expr;
158
159 ++tfound;
160
161 epos = vpos_end + 1;
162
163 exprvec.emplace_back( bool_expr_t{ bool_expression_t{ var_expr, expr }, truth::VARIABLE } );
164
165 } else if ( ec == '&' || ec == '|' || ec == '>' || ec == '!' ) {
166
167 std::string opstr;
168 if ( e+1 < exprstr.length() ) {
169 if ( exprstr[e+1] == '&' || exprstr[e+1] == '|' ) {
170 opstr += ec + exprstr[e+1];
171
172 epos = e+2;
173
174 ++tfound;
175 }
176 } else {
177 opstr += ec;
178
179 epos = e + 1;
180
181 ++tfound;
182 }
183
184 estr += opstr;
185
186 if ( opstr.length() == 2 ) {
187 BoolExpr opexpr = opstr;
188
189 exprvec.emplace_back( bool_expr_t{ bool_expression_t{ opexpr, expr }, truth::OPERATOR } );
190 } else if ( opstr.length() == 1 && opstr == "!" ) {
191 BoolExpr opexpr = opstr;
192
193 exprvec.emplace_back( bool_expr_t{ bool_expression_t{ opexpr, expr }, truth::NEGATION } );
194 }
195 }
196 }
197
198 return exprvec;
199}

202BoolType::expression_t BoolType::atomic_bracket_find( expr_t expr, size_t max_bracket_pairs ) {
203
204 BoolExpr* new_expr = nullptr;
205 std::string exprstr = expr;
206 size_t bpos = exprstr.find_first_of( "(" );
207
208 if ( bpos == exprstr.npos ) return { nullptr, expr };
209
210 atomic_expr_t* aet = new atomic_expr_t();
211
212 unsigned int bopen = 1;
213 unsigned int bmatch = 0;
214 while ( bopen > 0 && ( max_bracket_pairs > bmatch || max_bracket_pairs == 0 ) ) {
215
216 std::string bstr;
217 for ( unsigned int b = bpos; b < exprstr.length(); b++ ) {
218 char cpos = exprstr[b];
219
220 bstr += cpos;
221
222 if ( cpos == ')' && bopen == 1 ) { ++bmatch; break; }
223 else if ( cpos == ')' ) --bopen;
224 else if ( cpos == '(' ) ++bopen;
225 else if ( bmatch == max_bracket_pairs ) break;
226 }
227 bpos += exprstr.length();
228
229 new_expr = new BoolExpr( bstr );
230 atomic_term_t* att = new atomic_term_t{ *new_expr, new_expr->is() };
231 oneptr::queue<atomic_term_t>* qat = new oneptr::queue<atomic_term_t>();
232 qat->insert( qat->cbegin(), { att } );
233 bool_expression_t be{ "", "" };
234 term_t tt = BoolType::find_terms( *new_expr, be );
235 qat->append_range( std::ranges::subrange{ tt.first->second->begin(), tt.first->second->end() } );
236 atomic_terms_t* ats = new atomic_terms_t{ *new_expr, qat };
237 aet->insert( aet->cbegin(), { ats } );
238 }
239
240 return expression_t{ aet, *new_expr };
241}

254size_t BoolType::atomic_expr_count( expr_t expr ) {
255
256 std::string exprstr = expr;
257 unsigned int aec = 0;
258
259 for ( unsigned int epos = 0; epos < exprstr.length(); ) {
260 char e = exprstr[epos];
261
262 if ( e == '(' || e == '&' || e == '|' || e == ' ' ) {
263
264 if ( e == ' ' ) {
265 size_t tpos = exprstr.find_first_of( ' ', epos+1 );
266
267 if ( tpos != exprstr.npos ) ++aec;
268 else {
269 tpos = exprstr.find_last_of( ' ', exprstr.npos );
270
271 std::string term = exprstr.substr( tpos ).c_str();
272 if ( term != "&&" && term != "||" && term != "(" && term != ")" ) {
273 return aec;
274 } else ++aec;
275 }
276
277 epos = tpos;
278 continue;
279
280 } else ++aec;
281
282 ++aec;
283
284 epos = exprstr.find_first_of( ' ', epos );
285
286 } else epos++;
287 }
288
289 return aec;
290}

304BoolType::small_term_queue_t* BoolType::find_operator_in_expr( expr_t expr ) {
305
306 std::string str = expr;
307
308 small_term_queue_t* stqt = new small_term_queue_t();
309 expr_vec_t exprvec = BoolType::atomic_find( expr, 0 );
310
311 size_t pos = 0;
312 uint8_t setpos = 0;
313 small_term_t* stt = nullptr;
314 std::string v1, v2, o = "";
315 for ( bool_expr_t& bet : exprvec ) {
316 if ( bet.second == truth::VARIABLE ) {
317 if ( v1.length() == 0 ) {
318 v1 = bet.first.first;
319 ++setpos;
320 } else {
321 v2 = bet.first.first;
322 ++setpos;
323 }
324 } else if ( bet.second == truth::OPERATOR ) {
325 if ( o.length() == 0 ) {
326 o = bet.first.first;
327 ++setpos;
328 }
329 }
330
331 if ( v2.length() > 0 || ( o == "!" && v1.length() > 0 ) ) {
332 stt = new small_term_t{ { v1, v2 }, BoolOperator::is( o ) };
333 stqt->push_back( stt );
334
335 stt = nullptr;
336 v1 = "", v2 = "", o = "";
337 }
338
339 if ( setpos >= 3 ) {
340 setpos = 0;
341 stt = nullptr;
342 v1 = "", v2 = "", o = "";
343 }
344 }
345
346 return stqt;
347}

350BoolType::bindval_t* BoolType::find_first_expr_bind( expr_t expr, expr_t& restexpr ) {
351
352 std::string str = restexpr.length() > 0 ? restexpr : expr;
353
354 BoolExpr* e1, *e2 = nullptr;
355 BoolOperator::Type o = BoolOperator::UNKNOWN;
356 bool pre_op = true;
357 for ( unsigned int i = 0; i < str.length(); i++ ) {
358
359 char s = str[i];
360 std::string ops;
361 if ( i+1 < str.length() )
362 ops = str.substr( i, 2 );
363 else ops = str.substr( i, 1 );
364
365 if ( s == '&' || s == '|' || ops == "&&" || ops == "||" ) {
366 std::string opsstr = ops;
367 size_t gpos = str.find_first_of( ' ', i + opsstr.length() + 2 );
368 o = BoolOperator::is( str.substr( gpos ) );
369 pre_op = false;
370 } else if ( s == '!' ) {
371 o = BoolOperator::NEGATION;
372 pre_op = false;
373 } else if ( pre_op ) {
374 if ( s != ' ' ) *e1 += s;
375 } else {
376 if ( s != ' ' ) *e2 += s;
377 }
378 }
379
380 if ( e1->length() == 0 ) return nullptr;
381
382 if ( o == BoolOperator::NEGATION ) {
383 size_t pos_e1 = str.find_first_not_of( *e1 );
384 size_t pos_o = str.rfind( '!', pos_e1 );
385
386 std::string rst = str.substr( 0, pos_o - 1 );
387 rst += str.substr( pos_e1 );
388
389 restexpr.assign( rst );
390 } else {
391 size_t pos_e1 = str.find_first_of( *e1 );
392 size_t pos_e2 = str.find_first_not_of( *e2 );
393
394 std::string rst = str.substr( 0, pos_e1 );
395 rst += str.substr( pos_e2 );
396
397 restexpr.assign( rst );
398 }
399
400 return new bindval_t{ { *e1, *e2 }, o };
401}
