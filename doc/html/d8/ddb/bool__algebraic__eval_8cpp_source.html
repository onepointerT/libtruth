80void EvaluationExam::evaluate_all_groups() {
81
82 small_terms.clear();
83 evaluated = false;
84
85 for ( const oneptr::queue_element< AlgebraicBind >* qe : { evaluation_group.front(), evaluation_group.back() } ) {
86 std::vector< AlgebraicSmallTerm* > astvec_left = qe->get()->small_terms_left();
87 std::vector< AlgebraicSmallTerm* > astvec_right = qe->get()->small_terms_right();
88
89 auto add = [this](AlgebraicSmallTerm* term) {
90 this->small_terms.push_back( term );
91 };
92
93 for ( AlgebraicSmallTerm* ast : astvec_left ) { add( ast ); }
94 for ( AlgebraicSmallTerm* ast : astvec_right ) { add( ast ); }
95 }
96
97 evaluated = true;
98}

138bool Evaluator::eval( const BoolDict< BoolValue >& bd ) {
139 if ( expr.size() == 0 ) return false;
140
141 expr_t restexpr = "";
142 BoolType::bindval_queue_t* bqt = BoolType::find_bindvals( expr, restexpr );
143
144 for ( oneptr::queue_element< BoolType::bindval_t >* qe : *bqt ) {
145 groups.emplace_front(
146 new EvaluationGroup(
147 { new AlgebraicBind( *(qe->get()) ) }
148 )
149 );
150 }
151
152 BoolValue eval_group_result = BoolValue::None;
153 for ( oneptr::queue_element< EvaluationGroup >* qe_eg : groups ) {
154 qe_eg->get()->exam->evaluate_exam( bd );
155
156 BoolValue result, tmp = BoolValue::None;
157 oneptr::queue_element< AlgebraicSmallTerm >* ast_tmp;
158 for ( oneptr::queue_element< AlgebraicSmallTerm >* qe_ast : qe_eg->get()->exam->small_terms ) {
159 AlgebraicSmallTerm* ast = qe_ast->get();
160 if ( ast->term.first.first.length() == 0 ) {
161 result = ast->eval( bd, qe_ast->prev->get()->value );
162 } else if ( ast->term.first.second.length() == 0 ) {
163 ast_tmp = qe_ast;
164 } else {
165 result = ast->eval( bd );
166 }
167
168 if ( ast_tmp != nullptr ) {
169 result = ast->eval( bd, ast_tmp->get()->value );
170 ast_tmp = nullptr;
171 }
172
173 if ( ast->from == nullptr ) {
174 for ( oneptr::queue_element< AlgebraicBind >* qe_ab : *(qe_eg->get()) ) {
175 if ( qe_ab->get()->value.first.first.contains(ast->string())
176 || qe_ab->get()->value.first.second.contains(ast->string())
177 ) {
178 ast->from = qe_ab->get();
179 break;
180 }
181 }
182 }
183 if ( ast->from == nullptr ) {
184 qe_eg->get()->exam->value = BoolValue::None;
185 return false;
186 } else {
187 ast->from->push_evaluated_small_term( ast );
188 }
189
190 }
191
192 qe_eg->get()->exam->value = result;
193
194 qe_eg->get()->eval_if_complete( bd );
195
196 for ( oneptr::queue_element< AlgebraicBind >* qe_ab : *(qe_eg->get()) ) {
197 AlgebraicBind* ab = qe_ab->get();
198
199
200 }
201
202 qe_eg->get()->eval_if_complete( bd );
203 }
204
205 AtomicBoolTerm* abt = new AtomicBoolTerm( this->expr );
206 bool result = abt->eval( bd, this );
207 type.results[ this->expr.c_str() ] = result;
208
209 return result;
210}
